FILES=""
case "$M" in
  M4) FILES="$S"; python3 - <<'PY'

s=open(p).read()
a="      locked   <- writes.lockDefinition(pipelineId, preLockPlan.needsAdvisory, site => hooks.at(AfterLock(site)))\n"

  ;;
  M5) FILES="$P/PipelineStepRepository.scala"; sed -i '1092s/deletedRow.parentStepId, deletedRow.rootId, deletedRow.position/preRow.parentStepId, preRow.rootId, preRow.position/' $FILES ;;
  M6) FILES="$P/PipelineStepRepository.scala"; sed -i 's/private val MaxDeleteAttempts = 3/private val MaxDeleteAttempts = 1/' $FILES ;;

  SK3) FILES="$P/PipelineStepRepository.scala"; python3 - <<'PY'
p="backend/src/main/scala/com/helio/infrastructure/persistence/pipelines/PipelineStepRepository.scala"
s=open(p).read()
a="""              if (locks.active(LockSite.StepDeletePanelsFirst))
                PanelRowLocks.lockForDeletion(steps = preRemoval.tailDescendants + preRow.id)
                  .andThen(locks.lockPipelines(LockSite.StepDelete, Seq(preRow.pipelineId)))
"""
b="""              if (locks.active(LockSite.StepDeletePanelsFirst))
                locks.lockPipelines(LockSite.StepDelete, Seq(preRow.pipelineId))
                  .andThen(PanelRowLocks.lockForDeletion(steps = preRemoval.tailDescendants + preRow.id))
"""
assert a in s
open(p,"w").write(s.replace(a,b,1))
PY
  ;;
  SK5) FILES="$P/PipelineDraftWrites.scala"; sed -i 's/ORDER BY r.id FOR UPDATE OF r/ORDER BY r.id FOR KEY SHARE OF r/' $FILES ;;
  BASE) FILES="" ;;
esac
{ echo "MUTATION $M"; git diff -- $FILES; } > "$LOG.diff"
export HEL924_TEST_GROUP_CONCURRENCY=2 HELIO_TEST_JVM_XMX=4g
cd backend
CMD="testOnly"; for sp in "$@"; do CMD="$CMD com.helio.api.routes.pipelines.$sp"; done

echo "exit=$?" >> "$LOG"

[ -n "$FILES" ] && git checkout -- $FILES
git status --short | grep -v '^??' >> "$LOG"
