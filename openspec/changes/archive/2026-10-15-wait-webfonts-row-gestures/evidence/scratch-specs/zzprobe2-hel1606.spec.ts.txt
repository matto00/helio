import { expect, test } from "@playwright/test";
import { registerAndLogin } from "./support/auth";
import * as fs from "node:fs";
const OUT = process.env.PROBE_OUT ?? "/tmp/probe2.log";
for (const mode of ["abort-css", "abort-font", "delay-font"]) {
  test(`probe2 ${mode}`, async ({ page, request }) => {
    await registerAndLogin(page, request, { prefix: "hel1606q", displayName: "p", domain: "example.com", label: mode, isolate: true });
    for (let i = 0; i < 5; i++) await page.request.post("/api/data-sources", { data: { name: `P ${i}`, type: "static", columns: [{ name: "a", type: "integer" }], rows: [[1]] }, headers: { "X-Helio-Requested-With": "1" } });
    if (mode === "abort-css") await page.route(/fonts\.googleapis\.com/, (r) => r.abort());
    if (mode === "abort-font") await page.route(/fonts\.gstatic\.com/, (r) => r.abort());
    if (mode === "delay-font") await page.route(/fonts\.gstatic\.com/, async (r) => { await new Promise((x) => setTimeout(x, 1500)); await r.continue(); });
    await page.goto("/sources");
    const snap = async () => page.evaluate(() => {
      const link = document.querySelector('link[href*="fonts.googleapis.com/css2"]') as HTMLLinkElement;
      return { media: link.media, sheet: !!link.sheet, st: document.fonts.status, faces: [...document.fonts].filter((f) => f.status !== "unloaded").map((f) => `${f.family}:${f.status}`),
        res: performance.getEntriesByType("resource").filter((e) => /fonts\.g/.test(e.name)).map((e) => `${new URL(e.name).host} status=${(e as any).responseStatus} dur=${Math.round(e.duration)}`) };
    });
    const out: any[] = [];
    for (const d of [300, 1000, 2500]) { await page.waitForTimeout(d); out.push(await snap()); }
    fs.appendFileSync(OUT, JSON.stringify({ mode, out }) + "\n");
  });
}
