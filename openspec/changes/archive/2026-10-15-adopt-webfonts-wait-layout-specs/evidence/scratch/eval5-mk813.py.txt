src,dst=sys.argv[1],sys.argv[2]

i=t.index('test("surface 7: Connectors page"')
head,body=t[:i],t[i:]

    global body
    assert a in body, a
    body=body.replace(a,b,1)
head=head.replace('import { registerAndLogin } from "./support/auth";','''import { registerAndLogin } from "./support/auth";

async function boxes(label: string, loc: Locator) {
  const n = await loc.count();
  const out: string[] = [];

  console.log(`[probe813] ${label}: ${out.join(" ")}`);
}''',1)
P='          '
