#!/usr/bin/env bash
# summarize a run file: per after-wait dump held counts + fonts.ready outcome
for f in "$@"; do echo "--- $f"; sed 's/\x1b\[[0-9;]*[A-Za-z]//g' "$f" | grep -a "ready \|after-wait" | sed -E 's/fonts=\{"media":"([a-z]*)","status":"([a-z]*)","loading":([0-9]+).*/media=\1 status=\2 loading=\3/' | cut -c1-260; done
