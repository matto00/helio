// HEL-1809 MEASUREMENT HARNESS (not a gate): profiles one Pattern edit on a no-Output pipeline (case A, first edit) and a
// second edit (case B). Needs the instrumentation patch (instrumentation.patch) applied for the walk / edge logs.
// Env: H1809_OUT (file to append JSON lines), H1809_ROWS, H1809_COLS, H1809_RENDERS=1 (React fiber render attribution),
// HEL1808_CPU_THROTTLE (as the HEL-1683 spec), H1809_LABEL.
const CSRF = { "X-Helio-Requested-With": "1" };
const AUTH = { prefix: "hel1809", displayName: "HEL-1809", domain: "example.test" } as const;
const SAMPLE_URL = /\/api\/data-sources\/[^/]+\/sample(\?.*)?$/;
const ROWS = Number(process.env.H1809_ROWS ?? "2");
const COLS = Number(process.env.H1809_COLS ?? "1");
const RENDERS = process.env.H1809_RENDERS === "1";
const OUT = process.env.H1809_OUT ?? "/dev/null";
const LABEL = process.env.H1809_LABEL ?? "run";

async function post<T>(page: Page, url: string, data: unknown): Promise<T> {
  const res = await page.request.post(url, { data, headers: CSRF });
  expect(res.status(), `${url} ${await res.text()}`).toBeLessThan(300);
  return (await res.json()) as T;
}

async function seed(page: Page): Promise<string> {
  const columns = Array.from({ length: COLS }, (_, i) => ({
    name: i === 0 ? "text" : `c${i}`,
    type: "string",
  }));
  const cell = (r: number, c: number): string => (c === 0 ? `${"a".repeat(28)}!` : `v${r}-${c}`);
  let source: { id: string };
  if (ROWS > 400) {
    const csv =
      columns.map((c) => c.name).join(",") + "\n" +
      Array.from({ length: ROWS }, (_, r) => columns.map((_c, c) => cell(r, c)).join(",")).join("\n") + "\n";
    const res = await page.request.post("/api/data-sources", {
      multipart: { name: "hel1809-src", file: { name: "big.csv", mimeType: "text/csv", buffer: Buffer.from(csv) } },
      headers: CSRF,
    });
    expect(res.status(), await res.text()).toBeLessThan(300);
    source = (await res.json()) as { id: string };
  } else {
    source = await post<{ id: string }>(page, "/api/data-sources", {
      name: "hel1809-src",
      type: "static",
      columns,
      rows: Array.from({ length: ROWS }, (_, r) => columns.map((_c, c) => cell(r, c))),
    });
  }
  const pipeline = await post<{ id: string }>(page, "/api/pipelines", {
    name: "hel1809-pipe",
    roots: [{ sourceId: source.id }],
  });
  await post(page, `/api/pipelines/${pipeline.id}/steps`, {
    type: "stringops",
    config: { operation: "extractRegex", field: "text", outputColumn: "out", pattern: "(x)" },
  });
  if (process.env.H1809_OUTPUT === "1") {
    await post(page, `/api/pipelines/${pipeline.id}/outputs`, { kind: "table", name: "hel1809-out", config: {} });
  }
  await post(page, `/api/pipelines/${pipeline.id}/run`, {});
  return pipeline.id;
}
