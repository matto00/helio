import { expect, test } from "@playwright/test";

import { registerAndLogin } from "./support/auth";
import { AUTH, openPipeline, seedPipeline, sheet, SIZES } from "./support/hel1582Phone";

// PROBE HEL-1830 item 2 (committed as .txt only): tap (b)'s target, the "N fields" trigger INSIDE the open step sheet.
// (1) per animation frame, log the sheet trigger centres, the step-preview header's children, the sheet content and
//     StepCardFocused boxes and every scrolled ancestor's scrollTop; (2) log every steps/:id/preview request;
// (3) PROBE_MODE=stale: hold the preview response, record the trigger centre, release, wait for the reflow, then touch the
//     recorded (pre-reflow) point and record whether the panel opened and the element hit.
// Env: PROBE_SIZE (index into SIZES.slice(0,2)), PROBE_RATE (CDP CPU throttle), PROBE_MODE (log | stale),
//      PROBE_DELAY_MS (hold the preview response, log mode, this long after the sheet tap starts).
test.use({ isMobile: true, hasTouch: true, deviceScaleFactor: 2 });
const SIZE = SIZES.slice(0, 2)[Number(process.env.PROBE_SIZE ?? "0")];
const RATE = Number(process.env.PROBE_RATE ?? "1");
const MODE = process.env.PROBE_MODE ?? "log";
const DELAY = Number(process.env.PROBE_DELAY_MS ?? "-1");

test("probe: the sheet's N fields trigger over time and a stale tap", async ({ page, request }) => {
  test.setTimeout(180_000);
  await page.setViewportSize(SIZE);
  await registerAndLogin(page, request, { ...AUTH, label: "probe-t2b" });
  const pipelineId = await seedPipeline(page, false, true, true);
  const cdp = await page.context().newCDPSession(page);
  await cdp.send("Emulation.setCPUThrottlingRate", { rate: RATE });
  console.log(
    `STRESS-ACTIVE size=${SIZE.width}x${SIZE.height} cpu throttle ${RATE}x mode=${MODE} hold-delay=${DELAY}`,
  );
  const reqs: string[] = [];
  let t0req = 0;
  page.on("request", (r) => {
    if (/\/steps\/[^/]+\/preview/.test(r.url()) || /draft-preview/.test(r.url()))
      reqs.push(`${Math.round(performance.now() - t0req)}ms ${r.method()} ${new URL(r.url()).pathname}`);
  });
  page.on("response", (r) => {
    if (/\/steps\/[^/]+\/preview/.test(r.url()) || /draft-preview/.test(r.url()))
      reqs.push(`${Math.round(performance.now() - t0req)}ms <- ${r.status()} ${new URL(r.url()).pathname}`);
  });
  await page.addInitScript(() => {
    const w = window as unknown as { __log: string[]; __startLog: () => void };
    w.__log = [];
    w.__startLog = () => {
      const t0 = performance.now();
      let last = "";
      const tick = () => {
        const dlg = document.querySelector("dialog.bottom-sheet");
        if (dlg) {
          const box = (el: Element | null) => {
            if (!el) return "-";
            const r = el.getBoundingClientRect();
            return `(${Math.round(r.x)},${Math.round(r.y)} ${Math.round(r.width)}x${Math.round(r.height)})`;
          };
          const trigs = Array.from(dlg.querySelectorAll(".output-schema-disclosure__trigger")).map(
            (e) => {
              const r = e.getBoundingClientRect();
              return `@(${Math.round(r.x + r.width / 2)},${Math.round(r.y + r.height / 2)})`;
            },
          );
          const header = Array.from(dlg.querySelectorAll(".step-preview__header > *")).map(
            (e) => (e.className.split(" ")[0] || e.tagName.toLowerCase()) + box(e),
          );
          const content = dlg.querySelector(".mobile-step-sheet__content");
          const scrolled: string[] = [`dlg=${Math.round((dlg as HTMLElement).scrollTop)}`];
          for (let n: Element | null = content; n && n !== dlg; n = n.parentElement)
            if ((n as HTMLElement).scrollTop > 0)
              scrolled.push(`${n.className.split(" ")[0]}=${Math.round((n as HTMLElement).scrollTop)}`);
          const now = `triggers=[${trigs.join(",")}] header=[${header.join(" ")}] content=${box(content)} stepCard=${box(content?.firstElementChild ?? null)} preview=${box(dlg.querySelector(".step-preview"))} scrollTop{${scrolled.join(",")}}`;
          if (now !== last) {
            w.__log.push(`${Math.round(performance.now() - t0)}ms ${now}`);
            last = now;
          }
        }
        requestAnimationFrame(tick);
      };
      requestAnimationFrame(tick);
    };
  });
  await openPipeline(page, pipelineId, "dark");
  // tap (a) as the spec does, so the preview result for step 0 is whatever the spec's sequence leaves behind.
  await page.getByRole("tab", { name: "Preview" }).tap();
  const rows = page.getByRole("region", { name: /^Rows after step/ });
  await expect(rows).toBeVisible();
  await expect(rows.locator(".step-preview__count")).toBeVisible();
  await page.locator(".mobile-body .output-schema-disclosure__trigger").first().tap();
  await expect(page.getByRole("dialog", { name: "Output schema" })).toBeVisible();
  await page.keyboard.press("Escape");
  await expect(page.getByRole("dialog", { name: "Output schema" })).toBeHidden();
  // tap (b)'s own sequence
  await page.getByRole("tab", { name: "Steps" }).tap();
  let release: () => void = () => undefined;
  const held = new Promise<void>((resolve) => (release = resolve));
  if (MODE === "stale" || DELAY >= 0) {
    await page.route(/\/steps\/[^/]+\/preview/, async (route) => {
      await held;
      await route.continue();
    });
  }
  t0req = performance.now();
  await page.evaluate(() => (window as unknown as { __startLog: () => void }).__startLog());
  const open = page.locator(".spine-row__select").nth(0).tap();
  if (DELAY >= 0 && MODE !== "stale") {
    await new Promise((r) => setTimeout(r, DELAY));
    release();
  }
  await open;
  await expect(sheet(page)).toBeVisible();
  await sheet(page).evaluate((el) =>
    Promise.all(el.getAnimations().map((x) => x.finished.catch(() => undefined))),
  );
  const trigger = sheet(page).locator(".output-schema-disclosure__trigger").first();
  if (MODE === "stale") {
    await trigger.waitFor();
    const early = (await trigger.boundingBox())!;
    const [x, y] = [early.x + early.width / 2, early.y + early.height / 2];
    const countBefore = await sheet(page).locator(".step-preview__count").count();
    release();
    await expect(sheet(page).locator(".step-preview__count")).toBeVisible();
    await page.waitForTimeout(500);
    const late = (await trigger.boundingBox())!;
    const hit = await page.evaluate(
      ([px, py]) => {
        const el = document.elementFromPoint(px, py);
        return el ? `${el.tagName.toLowerCase()}.${el.className}` : "none";
      },
      [x, y],
    );
    console.log(
      `PROBE STALE count present before release: ${countBefore > 0}; early centre (${Math.round(x)},${Math.round(y)}) late centre (${Math.round(late.x + late.width / 2)},${Math.round(late.y + late.height / 2)}); element at the early point after reflow: ${hit}`,
    );
    await page.touchscreen.tap(x, y);
    let opened = true;
    try {
      await expect(page.getByRole("dialog", { name: "Output schema" })).toBeVisible({
        timeout: 3000,
      });
    } catch {
      opened = false;
    }
    console.log(`PROBE STALE tap at the pre-reflow point opened the panel: ${opened}`);
    // The fixed counterpart: the coordinate is taken AFTER the count is visible (the precondition the fix waits for).
    const settled = (await trigger.boundingBox())!;
    await page.touchscreen.tap(settled.x + settled.width / 2, settled.y + settled.height / 2);
    let openedAfter = true;
    try {
      await expect(page.getByRole("dialog", { name: "Output schema" })).toBeVisible({
        timeout: 3000,
      });
    } catch {
      openedAfter = false;
    }
    console.log(
      `PROBE FIXED tap at the settled centre (${Math.round(settled.x + settled.width / 2)},${Math.round(settled.y + settled.height / 2)}) after the count is visible opened the panel: ${openedAfter}`,
    );
  } else {
    await trigger.waitFor();
    await page.waitForTimeout(1500);
    await trigger.tap();
    let opened = true;
    try {
      await expect(page.getByRole("dialog", { name: "Output schema" })).toBeVisible({
        timeout: 3000,
      });
    } catch {
      opened = false;
    }
    console.log(`PROBE LOG tap (b) opened the panel: ${opened}`);
  }
  const log = await page.evaluate(() => (window as unknown as { __log: string[] }).__log);
  console.log(`PROBE-REQUESTS\n${reqs.join("\n")}\nPROBE-FRAMES (${log.length} distinct layouts)\n${log.join("\n")}`);
});
